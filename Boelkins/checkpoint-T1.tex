\section{Learning target T1, version 1}

Find the Taylor polynomials of a given function about a given center $a$.

\hrulefill

\begin{enumerate}[leftmargin=15pt]
	\item Suppose that $f(x)$ is some function, and that we know the following information:
	\[f(-1) = 3, \ f'(-1) = \frac{1}{2}, \ f''(-1) = -1, \, f'''(-1) = 0, \ f^{(4)}(-1) = \frac{1}{5} \]
	Determine $P_4(x)$, the degree 4 Taylor polynomial approximation to $y = f(x)$ centered at $a = -1$.

	\vfill

	\item Now suppose that $g(x) = \dfrac{1}{x} = x^{-1}$. Find the degree 2 Taylor polynomial approximation, $P_2(x)$, to $g(x)$, centered at $a=3$. Show your work!

	\vfill

\end{enumerate}
